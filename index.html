async function createPDF(data,shareMode){ const page = document.createElement("div"); page.style.width = "559px"; page.style.height = "794px"; page.style.background = "#f3f4f6"; page.style.padding = "20px"; page.style.fontFamily = "'Arial',sans-serif"; page.style.direction = "rtl"; page.style.display = "flex"; page.style.justifyContent = "center"; page.innerHTML = `
تقرير صيانة المصعد
Elevator Maintenance Report
SMOU Group
${box("رقم التقرير | Report No",data.reportNo)} ${box("التاريخ | Date",data.date)} ${box("الساعة | Time",data.time)} ${box("اسم العميل | Customer",data.customerName)} ${box("رقم العقد | Contract No",data.contractNo)} ${box("الإمارة | Emirate",data.emirate)} ${box("المنطقة | Area",data.area)} ${box("اسم الفني | Technician",data.technician)}
${section( "سبب الزيارة | Reason of Visit", data.visitReason )} ${section( "الإجراء المتبع | Action Taken", data.actionTaken )} ${section( "التوصيات الفنية | Technical Recommendation", data.recommendation )}
مبلغ مطلوب | Required Amount : ${data.requiredAmount || "0"}
توقيع العميل
Customer Signature
توقيع الفني
Technician Signature
`; document.body.appendChild(page); const canvas = await html2canvas(page,{ scale:2, backgroundColor:"#f3f4f6" }); const imgData = canvas.toDataURL("image/jpeg",1); const { jsPDF } = window.jspdf; const pdf = new jsPDF({ orientation:"portrait", unit:"px", format:[559,794] }); pdf.addImage( imgData, "JPEG", 0, 0, 559, 794 ); document.body.removeChild(page); const fileName = "SMOU-Report-" + data.reportNo + ".pdf"; if(shareMode){ const blob = pdf.output("blob"); const file = new File( [blob], fileName, {type:"application/pdf"} ); if( navigator.canShare && navigator.canShare({files:[file]}) ){ await navigator.share({ title:"SMOU Maintenance Report", files:[file] }); }else{ pdf.save(fileName); } }else{ pdf.save(fileName); } } function box(title,value){ return `
${title}
${value || ""}
`; } function section(title,value){ return `
${title}
${value || ""}
`; }
